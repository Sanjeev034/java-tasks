class Solution {
    public boolean rotateString(String s, String goal) {
        // If lengths don't match, goal can never be formed by shifting s
        if (s.length() != goal.length()) {
            return false;
        }
        
        // Concatenating s with itself contains all possible shifted variations of s
        String doubled = s + s;
        
        // Check if goal is a substring of s + s
        return doubled.contains(goal);
    }
}
