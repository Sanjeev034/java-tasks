import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.IOException;

public class Solution {

    public static long calculateStringSimilarity(String s) {
        int n = s.length();
        int[] z = new int[n];
        z[0] = n; // The full string's prefix similarity with itself is its total length

        int l = 0;
        int r = 0;

        for (int i = 1; i < n; i++) {
            if (i <= r) {
                z[i] = Math.min(r - i + 1, z[i - l]);
            }
            
            while (i + z[i] < n && s.charAt(z[i]) == s.charAt(i + z[i])) {
                z[i]++;
            }

            if (i + z[i] - 1 > r) {
                l = i;
                r = i + z[i] - 1;
            }
        }

        // Sum all similarity values
        long totalSimilarity = 0;
        for (int value : z) {
            totalSimilarity += value;
        }

        return totalSimilarity;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        
        String firstLine = br.readLine();
        if (firstLine == null || firstLine.trim().isEmpty()) {
            return;
        }
        
        int t = Integer.parseInt(firstLine.trim());

        StringBuilder output = new StringBuilder();
        for (int i = 0; i < t; i++) {
            String s = br.readLine().trim();
            output.append(calculateStringSimilarity(s)).append("\n");
        }

        System.out.print(output);
    }
}
